Absolutely! Creating a web application for flowcharts and Mermaid diagrams is not only possible but also a popular type of project. These apps typically combine visual diagramming (drag-and-drop nodes) with text-based diagramming (Mermaid's markdown-like syntax).
Here are the core components you'd need:

GoJS (Commercial, feature-rich)

Cytoscape.js (Graph theory focused)
Features: Drag-and-drop nodes, connection lines, zoom/pan, grouping, styling
2. Mermaid Integration
Mermaid.js: The official library to render diagrams from text
Features: Live preview, syntax highlighting (Monaco Editor or CodeMirror), export to PNG/SVG/PDF

// Core Graph Structure
interface FlowchartGraph {
  nodes: Node[];
  edges: Edge[];
  viewport: Viewport;
}

interface Node {
  id: string;
  type: 'process' | 'decision' | 'start' | 'end' | 'database';
  position: { x: number; y: number };
  dimensions: { width: number; height: number };
  data: {
    label: string;
    color?: string;
    icon?: string;
    metadata?: Record<string, any>;
  };
}

interface Edge {
  id: string;
  source: string;      // node id
  target: string;      // node id
  sourceHandle?: string; // which port (top/bottom/left/right)
  targetHandle?: string;
  type: 'straight' | 'bezier' | 'orthogonal';
  label?: string;
  style?: EdgeStyle;
}

interface Viewport {
  x: number;
  y: number;
  zoom: number;
}

Bidirectional Sync Implementation
TypeScript
Copy
// Core interfaces
interface CanonicalGraph {
  nodes: CanonicalNode[];
  edges: CanonicalEdge[];
}

// The "Single Source of Truth"
class DiagramEngine {
  private graph: CanonicalGraph;
  private visualEditor: VisualEditor;
  private mermaidEditor: MermaidEditor;
  private isSyncing: boolean = false;

  // Visual changes → Update Mermaid
  onVisualChange(newGraph: CanonicalGraph) {
    if (this.isSyncing) return;
    this.isSyncing = true;
    
    this.graph = newGraph;
    const mermaidCode = this.visualToMermaid(newGraph);
    this.mermaidEditor.setValue(mermaidCode);
    
    this.isSyncing = false;
  }

  // Mermaid changes → Update Visual
  onMermaidChange(code: string) {
    if (this.isSyncing) return;
    this.isSyncing = true;
    
    try {
      const parsed = mermaid.parse(code);
      this.graph = this.mermaidToVisual(parsed);
      this.visualEditor.render(this.graph);
    } catch (e) {
      // Show syntax error, don't update visual
      this.mermaidEditor.showError(e);
    }
    
    this.isSyncing = false;
  }

  // Converter: Visual → Mermaid text
  private visualToMermaid(graph: CanonicalGraph): string {
    const lines = ['flowchart TD'];
    
    for (const node of graph.nodes) {
      const shape = this.getMermaidShape(node.type, node.label);
      lines.push(`    ${node.id}${shape}`);
    }
    
    for (const edge of graph.edges) {
      const label = edge.label ? `|${edge.label}|` : '';
      lines.push(`    ${edge.source} -->${label} ${edge.target}`);
    }
    
    return lines.join('\n');
  }

  // Parser: Mermaid AST → Visual positions
  private mermaidToVisual(ast: MermaidAST): CanonicalGraph {
    // Use Dagre for automatic layout
    const g = new dagre.graphlib.Graph();
    g.setGraph({ rankdir: 'TB', nodesep: 50, ranksep: 80 });
    
    // Add nodes to Dagre
    for (const node of ast.nodes) {
      g.setNode(node.id, { 
        width: node.width, 
        height: node.height,
        label: node.text 
      });
    }
    
    // Add edges
    for (const edge of ast.edges) {
      g.setEdge(edge.from, edge.to);
    }
    
    // Calculate layout
    dagre.layout(g);
    
    // Extract positions
    const nodes = g.nodes().map(id => ({
      id,
      position: g.node(id),
      // ... other props
    }));
    
    return { nodes, edges: ast.edges };
  }
}
